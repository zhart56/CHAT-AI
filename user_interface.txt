def create_chat_input_interface(placeholder="Ketik pesan Anda atau unggah file..."):
    with gr.Row():
        user_input = gr.Textbox(show_label=False, placeholder=placeholder, scale=4, container=False)
        file_upload = gr.File(label=" 📁  Unggah File", file_count="single", scale=1)
    with gr.Row():
        send_button = gr.Button("Kirim Pesan", variant="primary", visible=True)
        send_vision = gr.Button(" 🖼️  Kirim Gambar+Teks (AI Vision)", variant="secondary", visible=False)
        send_ocr = gr.Button(" 📄  Baca Teks dari Gambar", variant="secondary", visible=False)
        send_doc = gr.Button(" 📄  Kirim Dokumen+Teks", variant="secondary", visible=False)
    
    def handle_file_upload(file_obj):
        if file_obj is None: return gr.update(visible=True), gr.update(visible=False), gr.update(visible=False), gr.update(visible=False)
        filename = file_obj.name.lower()
        if filename.endswith(('.png', '.jpg', '.jpeg', '.bmp', '.webp')):
            return gr.update(visible=False), gr.update(visible=True), gr.update(visible=True), gr.update(visible=False)
        elif filename.endswith(('.pdf', '.docx', '.txt')):
            return gr.update(visible=False), gr.update(visible=False), gr.update(visible=False), gr.update(visible=True)
        else: return gr.update(visible=True), gr.update(visible=False), gr.update(visible=False), gr.update(visible=False)
    
    file_upload.upload(handle_file_upload, file_upload, [send_button, send_vision, send_ocr, send_doc], show_progress="hidden")
    file_upload.clear(handle_file_upload, file_upload, [send_button, send_vision, send_ocr, send_doc], show_progress="hidden")
    
    return user_input, file_upload, send_button, send_vision, send_ocr, send_doc

def get_last_ai_response(chat_history):
    if not chat_history: return ""
    last_entry = chat_history[-1]
    if last_entry and last_entry.get('role') == 'assistant':
        content = last_entry.get('content', '')
        return content
    return ""

def build_rp_prompt(ai_persona, user_persona, pov_mode, memory_content):
    """Membangun system prompt final untuk roleplay."""
    pov_instructions = {
        "Sudut Pandang Kedua (Interaktif)": """
---
ATURAN SUDUT PANDANG KEDUA (UNTUK AI):
- PERAN ANDA: Anda adalah karakter yang dideskripsikan di atas.
- PERAN PENGGUNA: Pengguna adalah karakter yang dideskripsikan di atas.
- INTERAKSI: Bicaralah LANGSUNG KEPADA pengguna menggunakan kata 'Anda' atau 'kamu'. Deskripsikan dunia dan aksi dari sudut pandang orang pertama ('Saya melihat...', 'Aku merasakan...'). Fokus pada pengalaman interaktif pengguna.
""",
        "Sudut Pandang Ketiga (Naratif)": """
---
ATURAN SUDUT PANDANG KETIGA (UNTUK AI):
- PERAN ANDA: Anda adalah Narator atau Game Master (GM) yang juga mengontrol karakter AI.
- PERAN PENGGUNA: Pengguna mengontrol karakternya sendiri.
- INTERAKSI: Deskripsikan semua aksi (baik aksi Anda maupun aksi pengguna) dari sudut pandang orang ketiga. JANGAN PERNAH menggunakan kata 'Anda' atau 'kamu'. Sebut pengguna sebagai 'dia' atau dengan nama karakternya. Tulis balasan seperti sebuah cerita kolaboratif.
"""
    }
    
    full_persona = f"DESKRIPSI KARAKTER AI (PERAN ANDA):\n{ai_persona}\n\nDESKRIPSI KARAKTER PENGGUNA (PERAN SAYA):\n{user_persona}"
    final_prompt = full_persona + "\n" + pov_instructions.get(pov_mode, "")
    
    final_prompt_with_memory = build_prompt_with_memory(final_prompt, memory_content)
    
    return final_prompt_with_memory, full_persona
